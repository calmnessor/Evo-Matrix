### 通信拓扑

| 通道 | 协议 | 频率 | 延迟要求 | 数据量 |
|------|------|------|---------|--------|
| 相机 → 推理 | USB 3.0 / ROS Topic | 30Hz | <10ms | 640×480×3 × 3相机 ≈ 8MB/s |
| 推理 → 控制 | ROS Topic / 直连 | 50Hz | <5ms | 7D float32 × 50 = 1.4KB/s |
| 状态反馈 | ROS Topic / FCI 直读 | 500Hz | <2ms | 7D × 2 × 500 = 28KB/s |
| 安全监控 | 独立线程直连 | 500Hz | <2ms | — |
| 日志 | ROS 2 Bag | 按需 | — | GB 级/小时 |

from launch import LaunchDescription
from launch_ros.actions import Node, LifecycleNode
from launch.actions import DeclareLaunchArgument, ExecuteProcess

def generate_launch_description():
    return LaunchDescription([
        # ===== 参数 =====
        DeclareLaunchArgument('model_path', default_value='/models/openvla-7b'),
        DeclareLaunchArgument('task', default_value='pick_up_cup'),
        DeclareLaunchArgument('speed', default_value='0.5'),  # 速度倍率

        # ===== 相机节点 =====
        Node(
            package='realsense2_camera',
            executable='realsense2_camera_node',
            name='wrist_camera',
            parameters=[{
                'serial_no': '123456',
                'rgb.width': 640, 'rgb.height': 480,
                'depth.width': 640, 'depth.height': 480,
                'rgb.fps': 30, 'depth.fps': 30,
            }],
            remappings=[('/camera/color/image_raw', '/wrist/rgb')],
        ),
        Node(
            package='realsense2_camera',
            executable='realsense2_camera_node',
            name='front_camera',
            parameters=[{
                'serial_no': '789012',
                'rgb.width': 640, 'rgb.height': 480,
                'rgb.fps': 30,
            }],
            remappings=[('/camera/color/image_raw', '/front/rgb')],
        ),

        # ===== 机器人控制节点 =====
        Node(
            package='franka_ros2',
            executable='franka_robot_node',
            name='franka',
            parameters=[{'robot_ip': '172.16.0.2'}],
        ),

        # ===== VLA 推理节点 =====
        Node(
            package='vla_ros2',           # 自建的 VLA ROS 2 包
            executable='vla_inference',
            name='vla_inference',
            parameters=[{
                'model_path': '/models/openvla-7b',
                'control_frequency': 50.0,
                'action_horizon': 50,
                'speed_multiplier': 0.5,  # 首次部署用低速!
            }],
        ),

        # ===== 安全监控节点 =====
        Node(
            package='vla_ros2',
            executable='safety_monitor',
            name='safety_monitor',
            parameters=[{
                'max_force': 50.0,
                'max_velocity': 0.3,
                'workspace_bounds': [0.2, 0.8, -0.5, 0.5, 0.05, 0.7],
            }],
        ),

        # ===== 日志记录 =====
        ExecuteProcess(
            cmd=['ros2', 'bag', 'record',
                 '/wrist/rgb', '/front/rgb',
                 '/arm/joint_states',
                 '/vla/action',
                 '/safety/status',
                 '-o', 'vla_experiment'],
            output='screen',
        ),
    ])

### 延迟测量

```python
# 端到端延迟测量: 发一个脉冲动作 → 看机器人实际响应时间
import time
import numpy as np

def measure_e2e_latency(robot, camera, num_tests=50):
    """测量端到端延迟: 相机采集 → 推理 → 机器人动作"""
    latencies = []
    for i in range(num_tests):
        # 记录相机采集时间
        t_capture = camera.get_frame_timestamp()

        # VLA 推理 (实际场景下)
        t_infer_start = time.time()
        obs = camera.capture()
        action = vla_model.predict(obs)
        robot.send_action(action)
        t_action_sent = time.time()

        # 等待机器人的状态反馈确认动作已执行
        while robot.get_joint_positions() is None:
            pass
        t_confirmed = robot.get_state_timestamp()

        total_latency = t_confirmed - t_capture
        infer_latency = t_action_sent - t_infer_start
        control_latency = t_confirmed - t_action_sent

        latencies.append({
            "total_ms": total_latency * 1000,
            "infer_ms": infer_latency * 1000,
            "control_ms": control_latency * 1000,
        })

    # 报告
    total = np.array([l["total_ms"] for l in latencies])
    print(f"端到端延迟: {np.mean(total):.1f} ± {np.std(total):.1f} ms")
    print(f"  推理: {np.mean([l['infer_ms'] for l in latencies]):.1f} ms")
    print(f"  控制: {np.mean([l['control_ms'] for l in latencies]):.1f} ms")
    return latencies
```

## 4. 实验记录与日志

```python
import json
import datetime
from pathlib import Path

class ExperimentLogger:
    """VLA 真机实验记录器"""

    def __init__(self, log_dir="./experiments/"):
        self.log_dir = Path(log_dir)
        self.log_dir.mkdir(exist_ok=True)
        self.session_id = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
        self.log_file = self.log_dir / f"exp_{self.session_id}.jsonl"

    def log_config(self, config: dict):
        """记录实验配置"""
        config["type"] = "config"
        config["timestamp"] = datetime.datetime.now().isoformat()
        self._write(config)

    def log_episode(self, episode_id, result, metrics):
        """
        episode_id: str
        result: "success" / "failure" / "interrupted"
        metrics: {key: value} — 自定义指标
        """
        entry = {
            "type": "episode",
            "episode_id": episode_id,
            "result": result,
            "metrics": metrics,
            "timestamp": datetime.datetime.now().isoformat(),
        }
        self._write(entry)

    def log_error(self, error_type, message, stack_trace=""):
        """记录错误"""
        entry = {
            "type": "error",
            "error_type": error_type,
            "message": message,
            "stack_trace": stack_trace,
            "timestamp": datetime.datetime.now().isoformat(),
        }
        self._write(entry)

    def _write(self, entry):
        with open(self.log_file, "a") as f:
            f.write(json.dumps(entry) + "\n")

# 使用示例
logger = ExperimentLogger()
logger.log_config({
    "model": "openvla-7b",
    "task": "pick_up_cup",
    "speed": 0.5,
    "lighting": "office_normal",
})

for ep in range(10):
    try:
        result, metrics = run_episode(vla_model, robot, task)
        logger.log_episode(f"ep_{ep:03d}", result, metrics)
    except Exception as e:
        logger.log_error("episode_error", str(e), str(e.__traceback__))
```

## 5. 一键启动/停止脚本

```bash
#!/bin/bash
# vla_deploy.sh — 一键启动/停止 VLA 真机部署

case "$1" in
  start)
    echo "启动 VLA 部署..."
    # 1. 检查硬件
    echo "  检查相机连接..."
    rs-enumerate-devices | grep -q "D435i" || {
      echo "⚠ 相机未连接!"; exit 1; }

    # 2. 启动 ROS 2 节点
    echo "  启动 ROS 2 节点..."
    ros2 launch vla_ros2 vla_deployment_launch.py \
      model_path:=/models/openvla-7b \
      task:=pick_up_cup \
      speed:=0.5 &

    # 3. 等待就绪
    echo "  等待系统就绪..."
    sleep 5
    ros2 topic list | grep -q "/vla/action" || {
      echo "⚠ VLA 推理节点未启动!"; exit 1; }
    echo "✅ 系统就绪"
    ;;

  stop)
    echo "停止 VLA 部署..."
    pkill -f "vla_inference"
    pkill -f "ros2 bag record"
    pkill -f "franka_robot_node"
    echo "✅ 已停止"
    ;;

  status)
    echo "系统状态:"
    ros2 topic list | grep -E "(vla|camera|arm)"
    ros2 topic hz /vla/action --window 10
    ;;

  *)
    echo "用法: $0 {start|stop|status}"
    ;;
esac
```

## 6. 调试检查清单

```
□ 相机 OK?
  ros2 topic hz /wrist/rgb  → 应稳定在 30Hz
  rviz2 → 能看到画面

□ 机器人 OK?
  ros2 topic echo /arm/joint_states  → 数据在更新
  手动发关节命令 → 机器人能动

□ VLA OK?
  ros2 topic hz /vla/action  → 应稳定在 10-50Hz
  ros2 topic echo /vla/action  → 动作值在合理范围

□ 延迟 OK?
  measure_e2e_latency  → < 100ms

□ 标定 OK?
  放已知物体 → 视觉计算位置 vs 实际位置 → < 1cm

□ 安全 OK?
  safety_monitor 节点在运行
  急停按钮可用
```
