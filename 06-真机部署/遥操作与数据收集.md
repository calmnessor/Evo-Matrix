```python
class CameraSystem:
    """多相机系统"""

    def __init__(self, camera_configs):
        """
        camera_configs = [
            {"name": "wrist", "serial": "123456"},
            {"name": "front", "serial": "789012"},
        ]
        """
        self.pipelines = {}
        self.configs = camera_configs

    def start(self):
        for cfg in self.configs:
            pipeline = rs.pipeline()
            config = rs.config()
            config.enable_device(cfg["serial"])
            config.enable_stream(rs.stream.color, 640, 480, rs.format.bgr8, 30)
            pipeline.start(config)
            self.pipelines[cfg["name"]] = pipeline
        # 预热 (前几帧不稳定)
        for _ in range(30):
            self._get_frames()

    def _get_frames(self):
        frames = {}
        for name, pipeline in self.pipelines.items():
            f = pipeline.wait_for_frames()
            color = np.asanyarray(f.get_color_frame().get_data())
            frames[name] = color
        return frames

    def capture(self):
        """返回 {camera_name: bgr_image}"""
        return self._get_frames()

    def stop(self):
        for pipeline in self.pipelines.values():
            pipeline.stop()

@dataclass
class EpisodeConfig:
    max_steps: int = 500        # 最大步数 (50Hz × 10s = 500)
    control_freq: int = 50       # 控制频率 (Hz)
    task_name: str = "default"   # 任务名
    language_instruction: str = "complete the task"


class TeleoperationCollector:
    """遥操作数据收集器 — 核心类"""

    def __init__(self, robot, camera_system, sm_device,
                 output_dir="./collected_data/"):
        self.robot = robot
        self.cameras = camera_system
        self.sm = sm_device
        self.output_dir = output_dir
        self.recording = False
        self.episode_buffer = []

    def run_interactive(self, config: EpisodeConfig):
        """
        主循环: 等待开始录制 → 录制 → 保存
        """
        print(f"\n{'='*60}")
        print(f"任务: {config.task_name}")
        print(f"指令: {config.language_instruction}")
        print(f"按 SpaceMouse 左键 = 开始/停止录制")
        print(f"按 SpaceMouse 右键 = 切换夹爪")
        print(f"按 键盘 'q' = 退出")
        print(f"{'='*60}\n")

        dt = 1.0 / config.control_freq
        last_time = time.time()
        gripper_state = 0  # -1=开, 1=关, 0=未知

        running = True
        while running:
            # 1. 读输入
            delta, buttons = read_spacemouse(self.sm)

            # 左键: 切换录制状态
            if buttons[0] and not self._prev_button_0:
                self.recording = not self.recording
                if self.recording:
                    print(f"🔴 开始录制 Episode")
                    self.episode_buffer = []
                else:
                    print(f"⏹ 停止录制 ({len(self.episode_buffer)} 步)")
                    self._save_episode(config)
            self._prev_button_0 = buttons[0]

            # 右键: 切换夹爪
            if buttons[1] and not self._prev_button_1:
                gripper_open = self.robot.toggle_gripper()
                gripper_state = -1 if gripper_open else 1
                print(f"夹爪: {'开' if gripper_open else '关'}")
            self._prev_button_1 = buttons[1]

            # 2. 执行动作 (始终执行，不管是否录制)
            self.robot.move_ee_delta(delta)

            # 3. 获取观测
            images = self.cameras.capture()
            joint_pos = self.robot.get_joint_positions()

            # 4. 如果录制中 → 保存数据
            if self.recording:
                step_data = {
                    "observation": {
                        "images": images,  # {cam_name: (H,W,3) uint8}
                        "joint_positions": joint_pos,
                        "gripper_state": gripper_state,
                    },
                    "action": delta,  # 6-DoF 增量 (这就是 VLA 要学的!)
                    "action_gripper": gripper_state,
                    "timestamp": time.time(),
                }
                self.episode_buffer.append(step_data)

                if len(self.episode_buffer) % 50 == 0:
                    print(f"  已录制 {len(self.episode_buffer)} 步")

            # 5. 频率控制
            elapsed = time.time() - last_time
            if elapsed < dt:
                time.sleep(dt - elapsed)
            last_time = time.time()

            # 检查键盘退出
            key = cv2.waitKey(1) & 0xFF
            if key == ord('q'):
                running = False
                if self.recording:
                    self._save_episode(config)
                    self.recording = False

        print("遥操作结束")

    def _save_episode(self, config: EpisodeConfig):
        """保存 episode 为 HDF5 格式"""
        import os
        os.makedirs(self.output_dir, exist_ok=True)

        timestamp = time.strftime("%Y%m%d_%H%M%S")
        filename = f"{self.output_dir}/{config.task_name}_{timestamp}.h5"

        with h5py.File(filename, "w") as f:
            # 元数据
            f.attrs["task_name"] = config.task_name
            f.attrs["language_instruction"] = config.language_instruction
            f.attrs["num_steps"] = len(self.episode_buffer)
            f.attrs["control_frequency"] = config.control_freq

            # 每个 step 的观测和动作
            for i, step in enumerate(self.episode_buffer):
                grp = f.create_group(f"step_{i:04d}")

                # 图像: 每个相机单独存
                for cam_name, img in step["observation"]["images"].items():
                    grp.create_dataset(f"image_{cam_name}", data=img)

                # 关节位置
                grp.create_dataset("joint_positions",
                    data=step["observation"]["joint_positions"])

                # 动作
                grp.create_dataset("action_delta",
                    data=step["action"])
                grp.create_dataset("action_gripper",
                    data=[step["action_gripper"]])

                # 时间戳
                grp.create_dataset("timestamp",
                    data=[step["timestamp"]])

        print(f"✅ 已保存: {filename} ({len(self.episode_buffer)} 步)")
        self.episode_buffer = []


# ===== 5. 主程序 =====
```

```python
"""
VR 遥操作数据收集 (双手)
使用 Meta Quest 2 + Python + openvr / PyOpenVR

手部 → 末端映射:
  右手位置 → 右臂末端目标
  食指扳机 → 右臂夹爪 (0-1)
  左手位置 → 左臂末端目标
  左手扳机 → 左臂夹爪
"""
import numpy as np
import openvr
import time

class VRTeleoperator:
    def __init__(self, left_robot, right_robot):
        self.vr = openvr.init(openvr.VRApplication_Scene)
        self.left_robot = left_robot
        self.right_robot = right_robot

        # 获取手柄的 pose
        self.left_controller_idx = self._find_controller("left")
        self.right_controller_idx = self._find_controller("right")

    def _find_controller(self, role):
        """找到手柄设备"""
        for i in range(openvr.k_unMaxTrackedDeviceCount):
            device_class = self.vr.getTrackedDeviceClass(i)
            if device_class == openvr.TrackedDeviceClass_Controller:
                # 检查左右手
                role_str = self.vr.getTrackedDeviceProperty(
                    i, openvr.Prop_ControllerRoleHint_Int32
                )
                if "Left" in str(role_str) and role == "left":
                    return i
                if "Right" in str(role_str) and role == "right":
                    return i
        return -1

    def _get_controller_pose(self, idx):
        """获取手柄 4×4 位姿矩阵"""
        poses = self.vr.getDeviceToAbsoluteTrackingPose(
            openvr.TrackingUniverseStanding, 0,
            openvr.k_unMaxTrackedDeviceCount
        )
        if poses[idx].bPoseIsValid:
            return np.array(poses[idx].mDeviceToAbsoluteTracking).reshape(4, 4)
        return None

    def _get_trigger_value(self, idx):
        """读扳机值 (0-1) → 映射为夹爪开度"""
        state = self.vr.getControllerState(idx)
        # 扳机 = axis 1 (大多数 VR 手柄)
        return state[1].rAxis[1].x

    def run_teleop_loop(self, control_freq=50):
        dt = 1.0 / control_freq
        recording = False
        episode = []

        while True:
            # 读手柄
            left_pose = self._get_controller_pose(self.left_controller_idx)
            right_pose = self._get_controller_pose(self.right_controller_idx)
            left_grip = self._get_trigger_value(self.left_controller_idx)
            right_grip = self._get_trigger_value(self.right_controller_idx)

            if left_pose is not None:
                # 手柄位姿 → 机械臂末端目标
                self.left_robot.move_to_pose(left_pose[:3, 3], left_pose[:3, :3])
                self.left_robot.set_gripper(left_grip > 0.5)

            if right_pose is not None:
                self.right_robot.move_to_pose(right_pose[:3, 3], right_pose[:3, :3])
                self.right_robot.set_gripper(right_grip > 0.5)

            time.sleep(dt)
```

## 方案 3: ALOHA 式双臂主从控制 — 最高质量

### ALOHA 原理

```
ALOHA = A Low-cost Open-source Hardware system for bimanual teleoperation

核心思想: 用两套一模一样的机械臂 (Leader + Follower)
  Leader 臂 = 人类直接推着走 (零力控制/重力补偿)
  Follower 臂 = 复制 Leader 臂的关节角
  → 1:1 的动作映射，无延迟，精度最高

数据采集: 记录 Follower 臂的关节角 → 这就是训练数据
```

```python
"""
键盘控制机械臂
适合: 测试运动范围、调试、单轴运动
不适合: 收集正式训练数据 (精度不够)

import pygame
import numpy as np

class KeyboardTeleoperator:
    def __init__(self, robot):
        pygame.init()
        self.screen = pygame.display.set_mode((400, 300))
        self.robot = robot

        # 键盘映射
        self.key_map = {
            pygame.K_w:    [ 0,  0,  0.005, 0, 0, 0],  # +Z (上)
            pygame.K_s:    [ 0,  0, -0.005, 0, 0, 0],  # -Z (下)
            pygame.K_a:    [ 0,  0.005, 0, 0, 0, 0],  # +Y (左)
            pygame.K_d:    [ 0, -0.005, 0, 0, 0, 0],  # -Y (右)
            pygame.K_q:    [ 0.005, 0, 0, 0, 0, 0],  # +X (前)
            pygame.K_e:    [-0.005, 0, 0, 0, 0, 0],  # -X (后)
            pygame.K_SPACE: [0, 0, 0, 0, 0, 0],  # 特殊: 切换夹爪
        }

    def run(self):
        running = True
        speed = 1.0  # 速度倍率

        while running:
            keys_pressed = False
            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    running = False

                if event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_UP:
                        speed *= 1.5    # 加速
                    elif event.key == pygame.K_DOWN:
                        speed /= 1.5    # 减速

            # 读取按键
            keys = pygame.key.get_pressed()
            delta = np.zeros(6)
            for key, action in self.key_map.items():
                if keys[key]:
                    if key == pygame.K_SPACE:
                        self.robot.toggle_gripper()
                    else:
                        delta += np.array(action)
                    keys_pressed = True

            if keys_pressed and np.any(delta != 0):
                self.robot.move_ee_delta(delta * speed)

            pygame.time.delay(20)  # ~50Hz

        pygame.quit()
```

```python
def assess_data_quality(episode):
    """自动评估一条轨迹的质量"""
    issues = []

    # 1. 检查动作平滑度
    actions = np.array([s["action"] for s in episode])
    action_diff = np.diff(actions, axis=0)
    max_jerk = np.max(np.abs(np.diff(action_diff, axis=0)))
    if max_jerk > 0.5:
        issues.append(f"动作抖动严重 (max jerk={max_jerk:.3f})")

    # 2. 检查是否空走 (全是零动作)
    if np.all(np.abs(actions).sum(axis=1) < 0.001):
        issues.append("整个 episode 几乎没动 (可能是空录)")

    # 3. 检查关节速度是否超限
    joint_vel = np.diff([s["observation"]["joint_positions"] for s in episode], axis=0) / 0.02
    if np.any(np.abs(joint_vel) > 3.0):  # 3 rad/s
        issues.append("关节速度超过 3 rad/s")

    # 4. 检查 episode 长度
    if len(episode) < 20:
        issues.append(f"episode 太短 ({len(episode)} 步)")

    # 5. 检查任务是否完成 (看最后是否有抓取+提起动作)
    # 简化: 检查最后 10 步末端是否在物体上方
    # ... 更详细的检测需要物体跟踪

    if not issues:
        return "✅ 高质量", issues
    elif len(issues) <= 1:
        return "⚠ 可用", issues
    else:
        return "❌ 需要重录", issues
```

### 数据收集最佳实践

```
收集前:
  ✅ 确认标定准确 (手眼标定误差 < 1cm)
  ✅ 确认相机帧率稳定 (无掉帧)
  ✅ 清理场景，移除无关物体
  ✅ 物体放在自然位置 (不要太规整, VLA 需要多样性)
  ✅ 写好 episode 的元数据 (任务名、语言指令、难度)

收集中:
  ✅ 动作流畅、匀速 (不要忽快忽慢)
  ✅ 包含"接近→抓取→提起→移动→放置→松开"完整序列
  ✅ 夹爪在接触物体前先预对准 (不要"碰运气")
  ✅ 如果搞砸了: 从头重新录 (不要在一个失败的轨迹上继续)

收集后:
  ✅ 每一集后回放检查 (ros2 bag play)
  ✅ 确认没有关节超限/碰撞
  ✅ 确认图像清晰 (无过曝/模糊)
  ✅ 确认任务是否真的完成了 (看最后几帧)
  ✅ 给数据打标签: 任务名、成功/失败、质量评级
```

```
  □ 录制 5 条 "抓方块" 数据
  □ 评估数据质量 (用 assess_data_quality 函数)

  □ 设计数据收集计划 (什么任务、什么物体、多少条)

  □ 用收集的数据训练 BC 策略
  □ 在真机上 rollout 评估

```
