# 模型推理与实时优化

> 让 7B 参数的 VLA 模型在 50Hz 下实时跑在机器人上——从 PyTorch 到 TensorRT 的完整优化链路。

## 延迟预算

```
VLA 实时控制的延迟预算 (目标: 50Hz = 每步 20ms):

┌──────────┐  ┌──────────┐  ┌──────────┐  ┌──────────┐
│ 相机采集  │→│ VLA 推理 │→│ 后处理   │→│ 发送控制  │
│  ~5ms    │  │  ~10ms  │  │  ~2ms   │  │  ~3ms   │
└──────────┘  └──────────┘  └──────────┘  └──────────┘
                   ↑
              瓶颈在这!
              原始 PyTorch: 100-500ms (不够)
```

## 1. 基准测试 — 先知道瓶颈在哪

```python
import torch
import time
import numpy as np

def benchmark_model(model, batch_size=1, num_warmup=10, num_runs=100):
    """测试模型各环节耗时"""
    device = next(model.parameters()).device
    dummy_image = torch.randn(batch_size, 3, 224, 224, device=device)
    dummy_text = torch.randint(0, 1000, (batch_size, 64), device=device)

    # 预热
    for _ in range(num_warmup):
        _ = model(pixel_values=dummy_image, input_ids=dummy_text)

    # 计时
    torch.cuda.synchronize()
    times = []
    for _ in range(num_runs):
        start = time.perf_counter()
        _ = model(pixel_values=dummy_image, input_ids=dummy_text)
        torch.cuda.synchronize()
        times.append(time.perf_counter() - start)

    avg_ms = np.mean(times) * 1000
    std_ms = np.std(times) * 1000
    print(f"推理耗时: {avg_ms:.1f} ± {std_ms:.1f} ms")
    print(f"对应频率: {1000/avg_ms:.1f} Hz")

    # 分环节计时
    with torch.no_grad():
        # 视觉编码
        torch.cuda.synchronize()
        t0 = time.perf_counter()
        vision_features = model.vision_encoder(dummy_image)
        torch.cuda.synchronize()
        t1 = time.perf_counter()

        # 文本编码
        text_features = model.text_encoder(dummy_text)
        torch.cuda.synchronize()
        t2 = time.perf_counter()

        # 融合 + 动作生成
        full_output = model(vision_features, text_features)
        torch.cuda.synchronize()
        t3 = time.perf_counter()

    print(f"\n分步耗时:")
    print(f"  视觉编码: {(t1-t0)*1000:.1f} ms")
    print(f"  文本编码: {(t2-t1)*1000:.1f} ms")
    print(f"  融合+动作: {(t3-t2)*1000:.1f} ms")

    return times

# 实际部署时用 torch.profiler 更精确
def profile_model(model, dummy_inputs):
    """PyTorch Profiler 精确定位瓶颈"""
    with torch.profiler.profile(
        activities=[
            torch.profiler.ProfilerActivity.CPU,
            torch.profiler.ProfilerActivity.CUDA,
        ],
        schedule=torch.profiler.schedule(wait=2, warmup=2, active=5),
        on_trace_ready=torch.profiler.tensorboard_trace_handler('./logs'),
    ) as prof:
        for _ in range(10):
            with torch.no_grad():
                model(**dummy_inputs)
            prof.step()
    # 然后: tensorboard --logdir=./logs
```

## 2. 优化策略总览

| 方法 | 加速比 | 精度损失 | 实施难度 | 适合 |
|------|--------|---------|---------|------|
| **FP16/BF16** | ~2× | 几乎无 | ⭐ | 首选 |
| **INT8 量化** | ~3× | <1% | ⭐⭐ | 推理加速 |
| **INT4 量化** | ~4× | 2-5% | ⭐⭐ | 边缘部署 |
| **ONNX Runtime** | ~1.5× | 无 | ⭐⭐ | 跨平台 |
| **TensorRT** | ~3-5× | <1% | ⭐⭐⭐ | NVIDIA GPU ⭐ |
| **torch.compile** | ~1.3-2× | 无 | ⭐ | PyTorch 2.0+ |
| **KV Cache** | ~2× (第二步起) | 无 | ⭐⭐ | 自回归模型 |
| **模型蒸馏** | ~3-10× | 3-10% | ⭐⭐⭐⭐ | 终极方案 |
| **批次推理** | N× | 无 | ⭐ | 多机器人 |

## 3. FP16/BF16 — 最简单最有效

```python
import torch

# BF16 (推荐, 范围大, 训练稳定)
model = model.to(dtype=torch.bfloat16)

# FP16 (更快但可能溢出)
model = model.half()  # = .to(torch.float16)
```

## 4. TensorRT — NVIDIA GPU 最优推理

TensorRT 是 NVIDIA 的推理引擎，编译 PyTorch 模型为高度优化的 CUDA kernels。

### PyTorch → ONNX → TensorRT 流程

```python
# === 步骤 1: PyTorch → ONNX ===
import torch
import onnx

def export_to_onnx(model, output_path="vla_model.onnx"):
    """导出 VLA 的 vision 部分为 ONNX"""
    model.eval()

    # 定义动态 batch size
    dummy_image = torch.randn(1, 3, 224, 224, device="cuda")
    torch.onnx.export(
        model.vision_encoder,
        dummy_image,
        output_path,
        input_names=["image"],
        output_names=["vision_features"],
        dynamic_axes={
            "image": {0: "batch"},
            "vision_features": {0: "batch"},
        },
        opset_version=17,
        do_constant_folding=True,
    )
    print(f"✅ ONNX 已导出: {output_path}")


# === 步骤 2: ONNX → TensorRT ===
import tensorrt as trt

def build_tensorrt_engine(onnx_path, engine_path="vla.trt", fp16=True):
    """ONNX → TensorRT 引擎"""
    logger = trt.Logger(trt.Logger.WARNING)
    builder = trt.Builder(logger)
    network = builder.create_network(
        1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH)
    )
    parser = trt.OnnxParser(network, logger)

    with open(onnx_path, "rb") as f:
        if not parser.parse(f.read()):
            for i in range(parser.num_errors):
                print(f"❌ ONNX Parse Error: {parser.get_error(i)}")
            return None

    config = builder.create_builder_config()
    config.set_memory_pool_limit(trt.MemoryPoolType.WORKSPACE, 2 << 30)  # 2GB

    if fp16:
        config.set_flag(trt.BuilderFlag.FP16)

    # 构建引擎 (这一步最耗时, 10-30 分钟)
    print("正在构建 TensorRT 引擎...")
    engine = builder.build_serialized_network(network, config)

    with open(engine_path, "wb") as f:
        f.write(engine)
    print(f"✅ TensorRT 引擎已保存: {engine_path}")
    return engine


# === 步骤 3: TensorRT 推理 ===
import numpy as np

class TensorRTInference:
    def __init__(self, engine_path):
        self.logger = trt.Logger(trt.Logger.WARNING)
        self.runtime = trt.Runtime(self.logger)

        with open(engine_path, "rb") as f:
            self.engine = self.runtime.deserialize_cuda_engine(f.read())

        self.context = self.engine.create_execution_context()

    def infer(self, image_batch):
        """TensorRT 推理"""
        # 分配显存
        input_shape = (1, 3, 224, 224)
        output_shape = (1, 256, 1152)  # 视模型而定

        d_input = torch.empty(input_shape, dtype=torch.float16, device="cuda")
        d_output = torch.empty(output_shape, dtype=torch.float16, device="cuda")

        # 绑定
        self.context.set_tensor_address("image", d_input.data_ptr())
        self.context.set_tensor_address("vision_features", d_output.data_ptr())

        # 拷贝输入
        d_input.copy_(image_batch.half())

        # 执行推理
        self.context.execute_async_v3(torch.cuda.current_stream().cuda_stream)

        return d_output.float()  # 转回 FP32 供下游使用
```

### VLA 全流程 TensorRT 优化

```python
class OptimizedVLA:
    """全流程优化 VLA 推理:
    视觉: TensorRT
    LLM: 量化模型 + KV Cache
    动作: 保持 FM (Flow Matching 本身已经很快)
    """

    def __init__(self, vision_engine_path, llm_path):
        self.vision = TensorRTInference(vision_engine_path)
        self.llm = self._load_quantized_llm(llm_path)
        self.kv_cache = None  # KV Cache 复用

    def _load_quantized_llm(self, path):
        """加载量化 LLM"""
        from transformers import AutoModelForCausalLM, BitsAndBytesConfig
        bnb_config = BitsAndBytesConfig(
            load_in_4bit=True,
            bnb_4bit_compute_dtype=torch.bfloat16,
        )
        return AutoModelForCausalLM.from_pretrained(
            path,
            quantization_config=bnb_config,
            device_map="auto",
        )

    @torch.no_grad()
    def predict(self, image, text, proprio):
        # 1. 视觉编码 (TensorRT, ~5ms)
        vision_features = self.vision.infer(image)

        # 2. LLM 推理 (量化 + KV Cache, ~50ms)
        inputs = self._build_input(vision_features, text, proprio)
        if self.kv_cache is not None:
            # 复用 KV Cache (自回归只需推理新 token)
            outputs = self.llm(
                **inputs,
                past_key_values=self.kv_cache,
                use_cache=True,
            )
        else:
            outputs = self.llm(**inputs, use_cache=True)
        self.kv_cache = outputs.past_key_values

        # 3. 解码动作用 token (5ms)
        action_tokens = outputs.logits[:, -1, :].argmax(-1)
        return self._decode_actions(action_tokens)
```

## 6. 模型蒸馏 — 大模型教小模型

```python
"""
蒸馏思路:
  Teacher: OpenVLA-7B (大, 慢, 准确)
  Student: 小 Transformer (小, 快, 可以接受略低准确率)

  Loss = MSE(student预测, teacher预测)  ← 学大模型的"行为"
       + MSE(student预测, 真实动作)       ← 也学真实数据
"""
class DistillationTrainer:
    def __init__(self, teacher, student):
        self.teacher = teacher.eval()
        self.student = student

    def distill_step(self, batch, alpha=0.5):
        """
        alpha: 0=只学真实, 1=只学 teacher
        """
        image, text, true_action = batch

        # Teacher 推理 (不用梯度)
        with torch.no_grad():
            teacher_action = self.teacher(image, text)

        # Student 推理
        student_action = self.student(image, text)

        # 双重 loss
        loss_teacher = torch.nn.functional.mse_loss(student_action, teacher_action)
        loss_true = torch.nn.functional.mse_loss(student_action, true_action)
        loss = alpha * loss_teacher + (1 - alpha) * loss_true

        loss.backward()
        return loss.item()
```

## 7. 实时推理的异步架构

```python
"""
用 RTC 的思考方式: 推理和执行并行

┌─ 推理线程 ─┐     ┌─ 控制线程 ─┐
  推理 chunk n         执行 chunk n-1
  推理完成 → 发 buffer   buffer 用完 → 切换
  推理 chunk n+1       执行 chunk n
      ...                  ...

两个线程通过 action_buffer 同步
"""
import threading
import queue
import time

class AsyncVLAExecutor:
    def __init__(self, vla_model, robot, action_horizon=50):
        self.vla = vla_model
        self.robot = robot
        self.action_horizon = action_horizon
        self.buffer = queue.Queue(maxsize=2)  # 缓冲队列
        self.running = False

    def inference_thread(self):
        """推理线程: 持续推理，结果放 buffer"""
        while self.running:
            obs = self.robot.get_latest_obs()
            with torch.no_grad():
                action_chunk = self.vla.predict(obs)  # [H, act_dim]
            self.buffer.put(action_chunk)
            print(f"推理 → Buffer ({len(action_chunk)} 步)")

    def control_thread(self, control_freq=50):
        """控制线程: 从 buffer 拿动作执行"""
        dt = 1.0 / control_freq
        action_buffer = None
        buffer_idx = self.action_horizon

        while self.running:
            # 需要新 buffer?
            if buffer_idx >= len(action_buffer) if action_buffer is not None else True:
                action_buffer = self.buffer.get()  # 阻塞等待
                buffer_idx = 0
                print(f"切换 Buffer ({len(action_buffer)} 步)")

            # 执行
            action = action_buffer[buffer_idx]
            self.robot.send_action(action)
            buffer_idx += 1

            time.sleep(dt)

    def start(self):
        self.running = True
        t_infer = threading.Thread(target=self.inference_thread)
        t_control = threading.Thread(target=self.control_thread)
        t_infer.start()
        t_control.start()
        return t_infer, t_control
```
